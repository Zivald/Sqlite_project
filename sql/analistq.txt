---Какие категории дают больше всего повторных клиентов
WITH first_purchase AS (
    SELECT
        c.customer_unique,
        p.product_category,
        ROW_NUMBER() OVER (
            PARTITION BY c.customer_unique
            ORDER BY o.order_purchase_t
        ) AS rn
    FROM customers c
    JOIN orders o
        ON c.customer_id = o.customer_id
    JOIN order_items oi
        ON o.order_id = oi.order_id
    JOIN products p
        ON oi.product_id = p.product_id
    WHERE o.order_status = 'delivered'
),

first_category AS (
    SELECT
        customer_unique,
        product_category
    FROM first_purchase
    WHERE rn = 1
),
customer_orders AS (
    SELECT
        c.customer_unique,
        COUNT(DISTINCT o.order_id) AS orders_count
    FROM customers c
    JOIN orders o
        ON c.customer_id = o.customer_id
    WHERE o.order_status = 'delivered'
    GROUP BY c.customer_unique
)
SELECT COALESCE(t.product_category_1, fc.product_category) AS first_category,
COUNT(*) AS customers,
SUM(CASE
WHEN co.orders_count > 1 THEN 1 ELSE 0
END
) AS returned_customers,
ROUND(100.0 * SUM(CASE
WHEN co.orders_count > 1 THEN 1 ELSE 0
END
) / COUNT(*), 2
) AS repeat_rate
FROM first_category fc
JOIN customer_orders co
    ON fc.customer_unique = co.customer_unique
LEFT JOIN product_category_name_translation t
    ON fc.product_category = t.product_category
GROUP BY
    fc.product_category,
    t.product_category_1
HAVING COUNT(*) >= 50
ORDER BY repeat_rate DESC;


----Влияет ли оценка заказа на повторную покупку
WITH customer_reviews AS (
    SELECT
        c.customer_unique,
        AVG(r.review_score) AS avg_review_score,
        COUNT(DISTINCT o.order_id) AS orders_count
    FROM customers c
    JOIN orders o
        ON c.customer_id = o.customer_id
    JOIN order_reviews r
        ON o.order_id = r.order_id
    WHERE o.order_status = 'delivered'
    GROUP BY c.customer_unique
)
SELECT
    CASE
        WHEN avg_review_score >= 4.5
            THEN '4.5-5'
        WHEN avg_review_score >= 3.5
            THEN '3.5-4.49'
        WHEN avg_review_score >= 2.5
            THEN '2.5-3.49'
        ELSE '1-2.49'
    END AS review_group,
    COUNT(*) AS customers,
    ROUND(
        100.0 *
        SUM(
            CASE
                WHEN orders_count > 1 THEN 1
                ELSE 0
            END
        ) / COUNT(*),
        2
    ) AS repeat_rate
FROM customer_reviews
GROUP BY review_group
ORDER BY review_group DESC;
